cmake_minimum_required(VERSION 3.25)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "macOS deployment target")
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(diode909_m4l LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAX_SDK_BASE "" CACHE PATH "Path to Cycling '74 max-sdk-base")
if(NOT EXISTS "${MAX_SDK_BASE}/script/max-pretarget.cmake")
  message(FATAL_ERROR "Set MAX_SDK_BASE to a checkout of Cycling '74 max-sdk-base")
endif()
# The Max SDK builds externals with the static C runtime (/MT): the engine library must match.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(ROM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../roms" CACHE PATH "Folder with the decrypted TR-909 sample ROMs")
add_subdirectory(.. engine)
add_subdirectory(source/diode909_tilde)
